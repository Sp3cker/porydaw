void SettingsDialogCheckTest::init()

void SettingsDialogCheckTest::cleanup()
{
    QSettings settings;
    if (m_hadPcmMixer)
        settings.setValue(QStringLiteral("engine/pcmMixer"), m_pcmMixer);

        settings.remove(QStringLiteral("engine/pcmMixer"));
    settings.sync();
}
